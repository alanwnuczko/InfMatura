def algorytm2(s):
    n = len(s)

    r = [''] * n
    k = 0
    for i in range(n - 1, -1, -1):
        r[k] = s[i]
        k += 1

    ile = 0
    for i in range(n):
        if r[i] == s[i]:
            ile += 1
        else:
            break

    if ile == n:
        ile = ile // 2

    p = [''] * n
    for i in range(ile):
        p[i] = s[i]
        p[i + ile] = s[n + i - ile]

    return ile, p

print(algorytm2("abbcba"))

    k = 0
    n = len(s)

    r = [''] * n

        r[k] = s[i]
        k += 1

def dwa_kwadraty(n):
    for a in range(1, int(n**0.5) + 1):
        for b in range(1, int(n**0.5) + 1):
            if a**2 + b**2 == n:
                return a, b

print(dwa_kwadraty(29))
