4_1.py
licznik = 0

with open("liczby.txt") as f:
    for line in f:
        liczba = list(line.strip())

        licznik_0 = 0
        licznik_1 = 0
        for i in liczba:

            if i == "0":
                licznik_0 += 1
            else:
                licznik_1 += 1

        if licznik_0 > licznik_1:
            licznik += 1

print(licznik) # Wynik: 422
4_2.py
licznik_mod_2 = 0
licznik_mod_8 = 0

with open("liczby.txt") as f:
    for line in f:
        liczba = int(line.strip(), 2)

        if liczba % 2 == 0:
            licznik_mod_2 += 1

        if liczba % 8 == 0:
            licznik_mod_8 += 1

print(licznik_mod_2) # Wynik: 500
print(licznik_mod_8) # Wynik: 123
4_3.py
liczby = []
with open("liczby.txt") as f:
    for line in f:
        liczba = int(line.strip(), 2)
        liczby.append(liczba)

print(liczby.index(min(liczby)) + 1) # Wynik: 859
print(liczby.index(max(liczby)) + 1) # Wynik: 925
